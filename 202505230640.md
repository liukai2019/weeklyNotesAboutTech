```c++
#include <iostream>
#include <string>

// 抽象产品A（按钮）
class Button {
public:
    virtual void render() = 0;
    virtual ~Button() = default;
};

// 具体产品A1（Windows按钮）
class WindowsButton : public Button {
public:
    void render() override {
        std::cout << "Rendering a Windows button" << std::endl;
    }
};

// 具体产品A2（MacOS按钮）
class MacOSButton : public Button {
public:
    void render() override {
        std::cout << "Rendering a MacOS button" << std::endl;
    }
};

// 抽象产品B（文本框）
class TextBox {
public:
    virtual void render() = 0;
    virtual ~TextBox() = default;
};

// 具体产品B1（Windows文本框）
class WindowsTextBox : public TextBox {
public:
    void render() override {
        std::cout << "Rendering a Windows textbox" << std::endl;
    }
};

// 具体产品B2（MacOS文本框）
class MacOSTextBox : public TextBox {
public:
    void render() override {
        std::cout << "Rendering a MacOS textbox" << std::endl;
    }
};

// 抽象工厂
class GUIFactory {
public:
    virtual Button* createButton() = 0;
    virtual TextBox* createTextBox() = 0;
    virtual ~GUIFactory() = default;
};

// 具体工厂1（Windows风格）
class WindowsFactory : public GUIFactory {
public:
    Button* createButton() override {
        return new WindowsButton();
    }
    TextBox* createTextBox() override {
        return new WindowsTextBox();
    }
};

// 具体工厂2（MacOS风格）
class MacOSFactory : public GUIFactory {
public:
    Button* createButton() override {
        return new MacOSButton();
    }
    TextBox* createTextBox() override {
        return new MacOSTextBox();
    }
};

// 客户端调用
int main() {
    GUIFactory* factory;

    // 使用Windows风格
    factory = new WindowsFactory();
    Button* winButton = factory->createButton();
    TextBox* winTextBox = factory->createTextBox();
    winButton->render();
    winTextBox->render();

    // 使用MacOS风格
    factory = new MacOSFactory();
    Button* macButton = factory->createButton();
    TextBox* macTextBox = factory->createTextBox();
    macButton->render();
    macTextBox->render();

    // 释放资源
    delete winButton;
    delete winTextBox;
    delete macButton;
    delete macTextBox;
    delete factory;

    return 0;
}
```

```c++
    std::string m_cpu;
    std::string m_ram;
    std::string m_storage;

// 抽象构建器
class ComputerBuilder {
public:
    virtual ~ComputerBuilder() = default;
    virtual void buildCPU() = 0;
    virtual void buildRAM() = 0;
    virtual void buildStorage() = 0;
    virtual void buildGPU() = 0;  // 可选步骤
    virtual Computer getResult() = 0;
};

// 具体构建器：游戏电脑
class GamingComputerBuilder : public ComputerBuilder {
public:
    GamingComputerBuilder() { m_computer = Computer(); }

    void buildRAM() override { m_computer.setRAM("32GB DDR5"); }
    void buildStorage() override { m_computer.setStorage("1TB NVMe SSD"); }
    void buildGPU() override { m_computer.setGPU("NVIDIA RTX 4090"); }
    Computer getResult() override { return m_computer; }

private:
    Computer m_computer;
};

// 具体构建器：办公电脑
class OfficeComputerBuilder : public ComputerBuilder {

    OfficeComputerBuilder() { m_computer = Computer(); }

    Computer m_computer;

        m_builder->buildStorage();
```

```c++
#include <iostream>
#include <string>
#include <memory>

// 抽象原型类
class Prototype {
public:
    virtual ~Prototype() = default;
    virtual std::unique_ptr<Prototype> clone() const = 0;
    virtual void printInfo() const = 0;
};

// 具体原型类：敌人角色
class Enemy : public Prototype {
public:
    Enemy(const std::string& type, int health, int attackDamage)
        : m_type(type), m_health(health), m_attackDamage(attackDamage) {}

    // 关键：实现 clone 方法（深拷贝）
    std::unique_ptr<Prototype> clone() const override {
        return std::make_unique<Enemy>(*this);
    }

    void printInfo() const override {
        std::cout << "Enemy Type: " << m_type 
                  << ", Health: " << m_health 
                  << ", Attack: " << m_attackDamage << "\n";
    }

    // 修改内部状态（测试深拷贝）
    void setHealth(int health) { m_health = health; }

private:
    std::string m_type;
    int m_health;
    int m_attackDamage;
};

int main() {
    // 原始对象
    auto originalEnemy = std::make_unique<Enemy>("Dragon", 200, 50);
    std::cout << "Original: ";
    originalEnemy->printInfo();

    // 克隆对象
    auto clonedEnemy = originalEnemy->clone();
    std::cout << "Cloned: ";
    clonedEnemy->printInfo();

    // 修改克隆对象的状态（验证深拷贝）
    clonedEnemy->setHealth(150);
    std::cout << "After modifying clone:\n";
    std::cout << "Original: ";
    originalEnemy->printInfo();  // 原对象不受影响
    std::cout << "Cloned: ";
    clonedEnemy->printInfo();

    return 0;
}
```

组合模式
意图：将对象组合成树型结构以表示部分-整体的层次结构，composite，使得用户对单个对象和组合对象的使用具有一致性。

示例：

```c++
#include <iostream>
#include <string>
#include <vector>

// 组件接口
class FileSystemComponent {
public:
    virtual ~FileSystemComponent() = default;
    virtual void display(int depth = 0) const = 0;  // 显示名称和层级
    virtual void add(FileSystemComponent* component) {}  // 默认不实现（叶子节点无子节点）
    virtual void remove(FileSystemComponent* component) {}
};

// 叶子节点：文件
class File : public FileSystemComponent {
public:
    explicit File(const std::string& name) : m_name(name) {}

    void display(int depth) const override {
        std::cout << std::string(depth * 2, ' ') << "📄 " << m_name << std::endl;
    }

private:
    std::string m_name;
};

// 组合节点：文件夹
class Folder : public FileSystemComponent {
public:
    explicit Folder(const std::string& name) : m_name(name) {}

    void display(int depth) const override {
        std::cout << std::string(depth * 2, ' ') << "📁 " << m_name << std::endl;
        for (const auto& component : m_children) {
            component->display(depth + 1);  // 递归显示子节点
        }
    }

    void add(FileSystemComponent* component) override {
        m_children.push_back(component);
    }

    void remove(FileSystemComponent* component) override {
        auto it = std::find(m_children.begin(), m_children.end(), component);
        if (it != m_children.end()) {
            m_children.erase(it);
        }
    }

private:
    std::string m_name;
    std::vector<FileSystemComponent*> m_children;
};


int main() {
    // 创建文件和文件夹
    Folder root("根目录");
    Folder documents("文档");
    File resume("简历.pdf");
    File notes("笔记.txt");

    Folder images("图片");
    File photo1("photo1.jpg");
    File photo2("photo2.jpg");

    // 构建树形结构
    root.add(&documents);
    root.add(&images);

    documents.add(&resume);
    documents.add(&notes);

    images.add(&photo1);
    images.add(&photo2);

    // 递归显示整个结构
    root.display();

    return 0;
}
```

```c++
#include <iostream>
#include <string>
#include <memory>

// 组件接口：咖啡
class Coffee {
public:
    virtual ~Coffee() = default;
    virtual std::string getDescription() const = 0;
    virtual double cost() const = 0;
};

// 具体组件：黑咖啡
class BlackCoffee : public Coffee {
public:
    std::string getDescription() const override {
        return "黑咖啡";
    }
    double cost() const override {
        return 2.0;
    }
};

// 装饰器基类
class CoffeeDecorator : public Coffee {
protected:
    std::unique_ptr<Coffee> coffee;  // 持有被装饰的咖啡对象
public:
    explicit CoffeeDecorator(std::unique_ptr<Coffee> coffee)
        : coffee(std::move(coffee)) {}

    std::string getDescription() const override {
        return coffee->getDescription();
    }
    double cost() const override {
        return coffee->cost();
    }
};

// 具体装饰器：牛奶
class MilkDecorator : public CoffeeDecorator {
public:
    explicit MilkDecorator(std::unique_ptr<Coffee> coffee)
        : CoffeeDecorator(std::move(coffee)) {}

    std::string getDescription() const override {
        return CoffeeDecorator::getDescription() + " + 牛奶";
    }
    double cost() const override {
        return CoffeeDecorator::cost() + 0.5;
    }
};

// 具体装饰器：糖
class SugarDecorator : public CoffeeDecorator {
public:
    explicit SugarDecorator(std::unique_ptr<Coffee> coffee)
        : CoffeeDecorator(std::move(coffee)) {}

    std::string getDescription() const override {
        return CoffeeDecorator::getDescription() + " + 糖";
    }
    double cost() const override {
        return CoffeeDecorator::cost() + 0.3;
    }
};

// 客户端调用
int main() {
    // 创建基础咖啡
    auto coffee = std::make_unique<BlackCoffee>();
    std::cout << "基础咖啡: " << coffee->getDescription()
              << ", 价格: $" << coffee->cost() << std::endl;

    // 动态添加牛奶和糖
    auto milkCoffee = std::make_unique<MilkDecorator>(std::move(coffee));
    auto sweetMilkCoffee = std::make_unique<SugarDecorator>(std::move(milkCoffee));

    std::cout << "最终咖啡: " << sweetMilkCoffee->getDescription()
              << ", 价格: $" << sweetMilkCoffee->cost() << std::endl;

    return 0;
}
```

```c++
// 外观类：封装计算机启动流程
class ComputerFacade {
public:
    ComputerFacade() 
        : m_cpu(), m_memory(), m_hardDrive() {}

    void start() {
        // 1. CPU准备
        m_cpu.freeze();
        
        // 2. 从硬盘读取引导程序
        std::string bootData = m_hardDrive.readBootSector();
        
        // 3. 将引导程序加载到内存
        const long BOOT_ADDRESS = 0x7C00;
        m_memory.load(BOOT_ADDRESS, bootData);
        
        // 4. CPU跳转到引导地址并执行
        m_cpu.jump(BOOT_ADDRESS);
        m_cpu.execute();
    }

private:
    CPU m_cpu;
    Memory m_memory;
    HardDrive m_hardDrive;
};
```

```c++
#include <iostream>
#include <memory>
using namespace std;

// 抽象处理者
class Handler {
protected:
    shared_ptr<Handler> next;

public:
    void setNext(shared_ptr<Handler> nextHandler) {
        next = nextHandler;
    }

    virtual void handleRequest(int level) = 0;
    virtual ~Handler() = default;
};

// 具体处理者A：处理等级 <=1 的请求
class ConcreteHandlerA : public Handler {
public:
    void handleRequest(int level) override {
        if (level <= 1) {
            cout << "ConcreteHandlerA 处理了请求 level=" << level << endl;
        } else if (next) {
            next->handleRequest(level);
        } else {
            cout << "请求无法处理" << endl;
        }
    }
};

// 具体处理者B：处理等级 <=3 的请求
class ConcreteHandlerB : public Handler {
public:
    void handleRequest(int level) override {
        if (level <= 3) {
            cout << "ConcreteHandlerB 处理了请求 level=" << level << endl;
        } else if (next) {
            next->handleRequest(level);
        } else {
            cout << "请求无法处理" << endl;
        }
    }
};

// 具体处理者C：处理所有请求
class ConcreteHandlerC : public Handler {
public:
    void handleRequest(int level) override {
        cout << "ConcreteHandlerC 处理了请求 level=" << level << endl;
    }
};

// 客户端代码
int main() {
    shared_ptr<Handler> handlerA = make_shared<ConcreteHandlerA>();
    shared_ptr<Handler> handlerB = make_shared<ConcreteHandlerB>();
    shared_ptr<Handler> handlerC = make_shared<ConcreteHandlerC>();

    handlerA->setNext(handlerB);
    handlerB->setNext(handlerC);

    handlerA->handleRequest(1);
    handlerA->handleRequest(2);
    handlerA->handleRequest(4);

    return 0;
}
```

```c++
class Light {

class Command {

};

// 具体命令：打开灯
class LightOnCommand : public Command {

    Light* light;
public:
    LightOffCommand(Light* l) : light(l) {}

        light->off();
    }

class RemoteControl {
    Command* command;

        command = cmd;
    }
    void pressButton() {
        if (command)
            command->execute();
    }
};

    Light light;

    LightOnCommand lightOn(&light);
    LightOffCommand lightOff(&light);

    RemoteControl remote;

    remote.setCommand(&lightOn);
    remote.pressButton();  // 输出：Light is ON

    remote.setCommand(&lightOff);
    remote.pressButton();  // 输出：Light is OFF

    return 0;
```

中介者模式关键字：封装复杂的对象交互。


```c++
#include <iostream>
#include <string>
#include <vector>
#include <memory>
using namespace std;

// 前向声明
class User;

// 抽象中介者
class ChatRoomMediator {
public:
    virtual void showMessage(User* user, const string& message) = 0;
    virtual void addUser(User* user) = 0;
    virtual ~ChatRoomMediator() = default;
};

// 同事类
class User {
protected:
    string name;
    ChatRoomMediator* mediator;
public:
    User(string name, ChatRoomMediator* mediator)
        : name(name), mediator(mediator) {}

    string getName() const {
        return name;
    }

    void send(string message) {
        mediator->showMessage(this, message);
    }

    void receive(string message) {
        cout << "[" << name << "] 收到消息: " << message << endl;
    }
};

// 具体中介者
class ChatRoom : public ChatRoomMediator {
    vector<User*> users;
public:
    void addUser(User* user) override {
        users.push_back(user);
    }

    void showMessage(User* sender, const string& message) override {
        for (auto user : users) {
            if (user != sender) {
                user->receive(sender->getName() + " 说: " + message);
            }
        }
    }
};

// 客户端代码
int main() {
    ChatRoom chatroom;

    User user1("Alice", &chatroom);
    User user2("Bob", &chatroom);
    User user3("Charlie", &chatroom);

    chatroom.addUser(&user1);
    chatroom.addUser(&user2);
    chatroom.addUser(&user3);

    user1.send("大家好！");
    user2.send("你好，Alice！");
    user3.send("嘿，欢迎加入聊天室。");

    return 0;
}
```
